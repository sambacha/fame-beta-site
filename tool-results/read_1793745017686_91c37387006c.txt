   549→        <div className="flex justify-between gap-2 border-b border-stroke-dim/60 pb-1">
   550→          <dt className="text-content-faint">Utilization</dt>
   551→          <dd className="text-content-secondary">
   552→            {utilizationPct !== null ? `${utilizationPct.toFixed(0)}% of bound` : "—"}
   553→          </dd>
   554→        </div>
   555→      </dl>
   556→
   557→      <Dialog.Root open={dialogOpen} onOpenChange={setDialogOpen}>
   558→        <Dialog.Trigger
   559→          data-pulse={phase === "PARLEY_READY"}
   560→          className="mt-5 inline-flex min-h-11 items-center justify-center rounded-tremor-default bg-brand-green px-5 text-sm font-semibold text-surface-inset transition-colors hover:bg-brand-green-emphasis"
   561→        >
   562→          Confirm parametric order
   563→        </Dialog.Trigger>
   564→        <Dialog.Portal>
   565→          <Dialog.Backdrop className="fixed inset-0 z-50 bg-black/70" />
   566→          <Dialog.Popup className="fixed left-1/2 top-1/2 z-50 w-[calc(100%-2rem)] max-w-md -translate-x-1/2 -translate-y-1/2 rounded-tremor-default border border-stroke-dim bg-surface-base p-6 shadow-tremor-dropdown">
   567→            <Dialog.Title className="text-lg font-semibold text-content-primary">
   568→              Route parametric order
   569→            </Dialog.Title>
   570→            <Dialog.Description className="mt-2 text-sm text-content-secondary">
   571→              {legs} · {combinedApr.toFixed(1)}% combined APR · ρ {correlation.toFixed(2)}. The
   572→              margin engine will pre-clear both legs before the next slot.
   573→            </Dialog.Description>
   574→            <div className="mt-6 flex flex-wrap justify-end gap-3">
   575→              <Dialog.Close className="inline-flex min-h-11 items-center rounded-tremor-default border border-stroke-strong px-4 text-sm text-content-secondary transition-colors hover:text-content-primary">
   576→                Back
   577→              </Dialog.Close>
   578→              <button
   579→                type="button"
   580→                onClick={() => {
   581→                  onRoute()
   582→                  setDialogOpen(false)
   583→                }}
   584→                className="inline-flex min-h-11 items-center rounded-tremor-default bg-brand-green px-5 text-sm font-semibold text-surface-inset transition-colors hover:bg-brand-green-emphasis"
   585→              >
   586→                Route order
   587→              </button>
   588→            </div>
   589→          </Dialog.Popup>
   590→        </Dialog.Portal>
   591→      </Dialog.Root>
   592→    </div>
   593→  )

   596→/**
   597→ * Parametric Order Flow — a sequential, react-flow-style builder rendered as
   598→ * an accessible component pipeline: Node A (Select) → Node B (RadioGroup) →
   599→ * Node C (summary + confirm Dialog), connected by animated pseudo-element
   600→ * edges that respect prefers-reduced-motion.
   601→ */
   602→export function OrderFlowBuilder() {
   603→  const {
   604→    state,
   605→    tradeA,
   606→    leg,
   607→    orderData,
   608→    announcement,
   609→    handleNodeSelect,
   610→    handleAmountChange,
   611→    submit,
   612→    reset,
   613→  } = useOrderBuilder()
   614→  const { toast, dismiss } = useToast()
   615→  const reducedMotion = useReducedMotion()
   616→  const lastAttemptRef = useRef(0)
   617→  const lastAmountErrorRef = useRef<string | null>(null)
   618→
   619→  const legs: LegOption[] = tradeA ? (SECOND_LEGS[tradeA.id] ?? []) : []
   620→
   621→  // Non-destructive constraint toast when the margin engine rejects a pair.
   622→  useEffect(() => {
   623→    if (state.phase !== "INVALID_PAIR" || !state.invalidReason) return
   624→    if (lastAttemptRef.current === state.attempts) return
   625→    lastAttemptRef.current = state.attempts
   626→    const { id } = toast({
   627→      title: "Parley rejected by margin engine",
   628→      description: state.invalidReason,
   629→    })
   630→    const timer = window.setTimeout(() => dismiss(id), 6500)
   631→    return () => {
   632→      window.clearTimeout(timer)
   633→    }
   634→  }, [state.phase, state.invalidReason, state.attempts, toast, dismiss])
   635→
   636→  // Non-destructive toast when the ticket breaches the protocol's liquidity
   637→  // bounds — fires once per new constraint, never on every keystroke.
   638→  useEffect(() => {
   639→    const previous = lastAmountErrorRef.current
   640→    lastAmountErrorRef.current = state.amountError
   641→    if (!state.amountError || state.amountError === previous) return
   642→    const { id } = toast({
   643→      title: "Notional outside liquidity bounds",
   644→      description: state.amountError,
   645→    })
   646→    const timer = window.setTimeout(() => dismiss(id), 6500)
   647→    return () => {
   648→      window.clearTimeout(timer)
   649→    }
   650→  }, [state.amountError, toast, dismiss])
   651→
   652→  const nodeAStatus: NodeStatus = state.tradeAId ? "complete" : "active"
   653→  const nodeBStatus: NodeStatus =
   654→    state.phase === "INVALID_PAIR"
   655→      ? "error"
   656→      : state.phase === "TRADE_B_SELECTED"
   657→        ? "validating"
   658→        : state.phase === "IDLE"
   659→          ? "idle"
   660→          : state.phase === "TRADE_A_SELECTED"
   661→            ? "active"
   662→            : "complete"
   663→  const nodeCStatus: NodeStatus =
   664→    state.phase === "SUBMITTED" ? "complete" : state.phase === "PARLEY_READY" ? "active" : "idle"
   665→
   666→  const parley = tradeA && leg && state.phase === "PARLEY_READY" ? computeParlay(tradeA, leg) : null
   667→  const ticketBasis = orderData.summary?.basis === "ticket" ? orderData.summary : null
   668→
   669→  return (
   670→    <div className="rounded-tremor-default border border-stroke-dim bg-surface-inset/40 p-4 sm:p-6">
   671→      <ol className="grid list-none gap-4 p-0 lg:grid-cols-[1fr_1.5rem_1fr_1.5rem_1fr] lg:items-stretch lg:gap-0">
   672→        <FlowNode step="01" title="Primary trade" status={nodeAStatus}>
   673→          <p className="mb-4 text-sm text-content-secondary">
   674→            Pick the primary leg. The margin engine unlocks compatible pairs for it.
   675→          </p>
   676→          <TradeSelect
   677→            tradeAId={state.tradeAId}
   678→            onSelect={(id, label) => handleNodeSelect("tradeA", { id, label })}
   679→          />
   680→          {tradeA && (
   681→            <p className="mt-4 font-mono text-[11px] text-content-faint">
   682→              venue {tradeA.venue} · max notional ${(tradeA.maxNotionalUsd / 1_000_000).toFixed(1)}M
   683→              · {RISK_LABELS[tradeA.risk]} risk
   684→            </p>
   685→          )}
   686→        </FlowNode>
   687→
   688→        <FlowEdge active={Boolean(state.tradeAId)} />
   689→
   690→        <FlowNode step="02" title="Second leg" status={nodeBStatus}>
   691→          <p className="mb-4 text-sm text-content-secondary">
   692→            {tradeA
   693→              ? "Instruments below are pre-filtered for a valid parametric parley."
   694→              : "Unlocks after the primary trade is selected."}
   695→          </p>
   696→          <RadioGroup
   697→            aria-label="Second leg selection"
   698→            aria-invalid={state.phase === "INVALID_PAIR" ? "true" : undefined}
   699→            value={state.legId ?? null}
   700→            onValueChange={(value) => {
   701→              if (typeof value === "string") {
   702→                const match = legs.find((legOption) => legOption.id === value)
   703→                if (match) handleNodeSelect("legB", { id: match.id, label: match.label })
   704→              }
   705→            }}
   706→            className="grid gap-2"
   707→          >
   708→            {legs.length > 0 ? (
   709→              legs.map((legOption) => (
   710→                <LegRadio
   711→                  key={legOption.id}
   712→                  leg={legOption}
   713→                  name={tradeA?.label ?? "—"}
   714→                  disabled={!tradeA}
   715→                />
   716→              ))
   717→            ) : (
   718→              <p
   719→                aria-live="polite"
   720→                className="rounded-tremor-default border border-dashed border-stroke-dim p-4 text-sm text-content-faint"
   721→              >
   722→                Awaiting primary trade…
   723→              </p>
   724→            )}
   725→          </RadioGroup>
   726→          {state.phase === "INVALID_PAIR" && state.invalidReason && (
   727→            <p role="alert" className="mt-3 text-xs leading-relaxed text-brand-coral">
   728→              {state.invalidReason}
   729→            </p>
   730→          )}
   731→        </FlowNode>
   732→
   733→        <FlowEdge active={state.phase === "PARLEY_READY" || state.phase === "SUBMITTED"} />
   734→
   735→        <FlowNode step="03" title="Parley summary" status={nodeCStatus}>
   736→          <SummaryPanel
   737→            phase={state.phase}
   738→            ticket={state.ticket}
   739→            legs={parley?.legs ?? ""}
   740→            correlation={leg?.correlation ?? 0}
   741→            combinedApr={parley?.combinedApr ?? 0}
   742→            notionalUsd={ticketBasis?.notionalUsd ?? null}
   743→            utilizationPct={ticketBasis?.utilizationPct ?? null}
   744→            onRoute={submit}
   745→          />
   746→          {state.phase === "SUBMITTED" && (
   747→            <button
   748→              type="button"
   749→              onClick={reset}
   750→              className="mt-5 inline-flex min-h-11 items-center rounded-tremor-default border border-stroke-strong px-4 font-mono text-sm text-content-secondary transition-colors hover:border-brand-green hover:text-brand-green"
   751→            >
   752→              Build another parley ↺
   753→            </button>
   754→          )}
   755→        </FlowNode>
   756→      </ol>
   757→
   758→      {/* Amount → dependent second leg → payoff projection, one synchronized surface. */}
   759→      <OrderConfigPanel
   760→        tradeA={tradeA}
   761→        leg={leg}
   762→        orderData={orderData}
   763→        amountRaw={state.amountRaw}
   764→        amountError={state.amountError}
   765→        reducedMotion={reducedMotion}
   766→        onAmountChange={handleAmountChange}
   767→      />
   768→
   769→      <div role="status" aria-live="polite" className="sr-only">
   770→        {announcement}
   771→      </div>
   772→    </div>
   773→  )
   774→}
